3. Target Architecture

Merchant App→Private Storage→AI Edge Function→Provider Adapter→Draft + Confidence→Human Review→Atomic Commit RPC

Trust boundaries

  • Mobile ไม่มี provider API key หรือ service-role key
  • Function ตรวจ JWT, merchant ownership, object prefix และ request budget
  • ข้อความในภาพถือเป็นข้อมูล ไม่ใช่คำสั่ง prompt
  • provider response ต้องผ่าน strict schema ก่อน insert draft

Failure behavior

  • provider timeout → job เป็น failed_retryable; เมนูเดิมไม่เปลี่ยน
  • บางรูปอ่านไม่ได้ → คืนผลบางส่วนพร้อม warning
  • commit ซ้ำ → คืนผลเดิมจาก idempotency guard
  • budget เกิน → block ก่อนเรียก provider และบันทึก usage event

4.2 Provider-neutral types

export type ConfidenceLevel = "high" | "review" | "unreadable";

export interface AiMenuDraftItem {
  sourceIndex: number;
  name: string;
  price: number | null;
  categoryName: string | null;
  description: string | null;
  options: Array<{ name: string; priceDelta: number }>;
  confidence: { name: number; price: number; category: number };
  level: ConfidenceLevel;
  warnings: string[];
}

export interface MenuExtractionProvider {
  extract(input: {
    images: Uint8Array[];
    locale: "th-TH";
    requestId: string;
  }): Promise<{ items: AiMenuDraftItem[]; usage: ProviderUsage }>;
}
